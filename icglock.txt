def sns_notification(username):
    
    client = boto3.client('sns',region_name)
    subject = 'User Locked:' + username
    message = 'User' + username + 'has been locked.'
    status=client.publish(TargetArn='arn:aws:sns:eu-west-1:734006704080:V3DBA_Team',Message=message,Subject=subject)
    print(status)
    

def lambda_handler(event, context):
 
    dsn = cx_Oracle.makedsn(ENDPOINT, '1521',SID)
    conn = cx_Oracle.connect('master',PASSWORD,dsn)
    print("Oracle DB version ::::: = " + conn.version)
    curs = conn.cursor()
    curs.callproc("dbms_output.enable")
    
    sqlCode = """
    Declare
       lv_open_accounts   VARCHAR2 (2000);
       n_cnt              NUMBER := 0;
       FUNCTION f_get_open_db_accounts
          RETURN VARCHAR2
       IS
          lv_open_accounts   VARCHAR2 (2000);
          n_cnt              NUMBER := 0;
       BEGIN
          BEGIN
             FOR i
             IN (SELECT   username
                          ,account_status
                   FROM   dba_users
                    --WHERE   account_status = 'LOCKED'
                    WHERE   account_status = 'OPEN'
                          AND username NOT IN
                                   ('SYS',
                                    'SYSTEM',
                                    'KDELAHUNTY',
                                    'SVEERAMANENI',
                                    'MRAHMAN',
                                    'LNAGARAJA',
                                    'MCODY',
                                    'DBSNMP',
                                    'MASTER',
                                    'BENDEV',
                                    'AUPADHYAY',
                                    'RDSADMIN',
                                    'GENHOFFER',
                                    'RNBY',
				    'PERFSTAT'))
             LOOP
                IF n_cnt = 0
                THEN
                   lv_open_accounts := i.username;
                ELSE
                   lv_open_accounts := lv_open_accounts || chr(10) || i.username;
                END IF;
    
                n_cnt := n_cnt + 1;
    
                   execute immediate 'alter user '||i.username||' account lock'||'';
             END LOOP;
          EXCEPTION
             WHEN OTHERS
             THEN
                lv_open_accounts := NULL;
          END;
    
          RETURN trim(lv_open_accounts);
       END f_get_open_db_accounts;
    
    BEGIN
    
            lv_open_accounts := f_get_open_db_accounts;
            dbms_output.put_line(lv_open_accounts);
    
    END;
    """
    
    curs.execute(sqlCode)
    
    statusVar = curs.var(cx_Oracle.NUMBER)
    lineVar = curs.var(cx_Oracle.STRING)
    body = ''
    print(f'statusVar {statusVar}' )
    print(f'lineVar {lineVar}' )
    while True:
      curs.callproc("dbms_output.get_line", (lineVar, statusVar))
      if statusVar.getvalue() != 0:
        break
      #print lineVar.getvalue()
      body = body+str(lineVar.getvalue())+'\n'
    if str(body).strip() != 'None' :
        print('Locked Users are ===>',body)
        sns_notification(body)
    else:
        print ('No users found to be locked')
    
    curs.close()
    conn.close()
